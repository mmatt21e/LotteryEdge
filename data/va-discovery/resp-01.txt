<script>
/*** Promo Carousel (forward-only loop, temp-clone wrap, 5s autoplay) ***/
document.addEventListener('DOMContentLoaded', function () {
  const carousel = document.getElementById('promoCarousel');
  if (!carousel) return;

  const track = carousel.querySelector('.promo-carousel');
  if (!track) return;

  let slides = Array.from(track.querySelectorAll('.item'));
  const indicators = Array.from(document.querySelectorAll('.promo-carousel-controls .indicator'));
  const pauseBtn = document.getElementById('pauseCarouselBtn');
  const playBtn = document.getElementById('playCarouselBtn');

  const SLIDE_DELAY = 6000;
  const TRANSITION_MS = 600; // must match CSS transition: 0.6s
  let currentIndex = Math.max(0, slides.findIndex(s => s.classList.contains('active')));
  let timer = null;
  let isPlaying = true;
  let isWrapping = false;
  
  swapCarouselImages();

  function slideWidth() {
    return carousel.getBoundingClientRect().width;
  }


  function setIndicators(activeIndex) {
    indicators.forEach((dot, i) => {
      dot.classList.toggle('active', i === activeIndex);
      dot.src = dot.src.replace(
        /carousel-(active|inactive)\.svg/i,
        `carousel-${i === activeIndex ? 'active' : 'inactive'}.svg`
      );
    });
  }

  function setActiveSlide(activeIndex) {
    slides.forEach((s, i) => s.classList.toggle('active', i === activeIndex));
  }

  function setTransformPx(px, animate) {
    track.style.transition = animate ? 'transform 0.6s ease-in-out' : 'none';
    track.style.transform = `translate3d(-${px}px, 0, 0)`;
  }

  function goTo(index, animate = true) {
    currentIndex = index;
    setActiveSlide(currentIndex);
    setIndicators(currentIndex);
    setTransformPx(currentIndex * slideWidth(), animate);
  }

  function cleanupTempClone() {
    const temp = track.querySelector('.tmn-temp-clone');
    if (temp) temp.remove();
    slides = Array.from(track.querySelectorAll('.item'));
  }

  function next() {
    if (isWrapping) return;

    const realTotal = slides.length; // IMPORTANT: this is real slides only (no lingering clone)
    const lastRealIndex = realTotal - 1;

    // Normal advance
    if (currentIndex < lastRealIndex) {
      goTo(currentIndex + 1, true);
      return;
    }
	

	function forceReflow(el) {
	  el.offsetHeight; // forces layout recalculation
	}
	

    // Wrap: last → first, forward-only, no reverse snap
    isWrapping = true;

    // Ensure no previous temp clone exists
    cleanupTempClone();

    // Create temp clone of first slide, append to end
    const firstClone = slides[0].cloneNode(true);
    firstClone.classList.remove('active', 'slide-in');
    firstClone.classList.add('tmn-temp-clone');
    track.appendChild(firstClone);

    // Rebuild slides list INCLUDING the temp clone (clone is at the end now)
    const slidesWithClone = Array.from(track.querySelectorAll('.item'));
    const cloneIndex = slidesWithClone.length - 1;

    // During this transition, user wants indicator #1 selected (index 0)
    setIndicators(0);

    // Animate to the clone (this is a forward move left)
    setTransformPx(cloneIndex * slideWidth(), true);

    // After the transition completes, instantly snap back to real first without animation
    setTimeout(() => {
      // Remove clone so it never “duplicates” after cycle
      cleanupTempClone();

      // Snap to real first, no animation
      currentIndex = 0;
      setActiveSlide(0);
      setIndicators(0);
      setTransformPx(0, false);

      // Force reflow then restore transition (prevents next animation from being skipped)
      track.offsetHeight;
      track.style.transition = 'transform 0.6s ease-in-out';

      isWrapping = false;
    }, TRANSITION_MS);
  }

  function start() {
    stop();
    timer = setInterval(next, SLIDE_DELAY);
    isPlaying = true;
    if (pauseBtn) pauseBtn.style.display = 'inline';
    if (playBtn) playBtn.style.display = 'none';
  }

  function stop() {
    if (timer) clearInterval(timer);
    timer = null;
    isPlaying = false;
    if (pauseBtn) pauseBtn.style.display = 'none';
    if (playBtn) playBtn.style.display = 'inline';
  }
  


	function swapCarouselImages() {
	  //const isMobile = window.matchMedia('(max-width: 800px)').matches;
	  const isMobile = (window.innerWidth <= 800);

	  document.querySelectorAll('#promoCarousel .item').forEach(item => {
		const imgLarge = item.querySelector('.img-large');
		const imgSmall = item.querySelector('.img-small');

		if (!imgLarge || !imgSmall) return;

		imgLarge.hidden = isMobile;
		imgSmall.hidden = !isMobile;
	  });
	}


	function updateCarousel(animate = true) {
	  track.style.transition = animate ? 'transform 0.6s ease-in-out' : 'none';
	  track.style.transform =
		`translateX(-${currentIndex * slideWidth()}px)`;
	}

  

  // Indicator clicks (jump directly)
  indicators.forEach((dot, i) => {
    dot.addEventListener('click', () => {
      if (isWrapping) return;
      cleanupTempClone();
      goTo(i, true);
      if (isPlaying) start();
    });
  });

  // Pause / Play
  if (pauseBtn) pauseBtn.addEventListener('click', stop);
  if (playBtn) playBtn.addEventListener('click', start);

/*
	document.addEventListener('DOMContentLoaded', () => {
	  swapCarouselImages();
	});
*/


  // Keep alignment on resize
  /*
  window.addEventListener('resize', () => {
    cleanupTempClone();
    setTransformPx(currentIndex * slideWidth(), false);
    track.offsetHeight;
    track.style.transition = 'transform 0.6s ease-in-out';
	swapCarouselImages();
  });
  */
  

	let resizeTimer;


	window.addEventListener('resize', () => {
		cleanupTempClone();
		setTransformPx(currentIndex * slideWidth(), false);
		track.offsetHeight;
		track.style.transition = 'transform 0.6s ease-in-out';
			// Swap images
			swapCarouselImages();
			
		  clearTimeout(resizeTimer);
		  resizeTimer = setTimeout(() => {

			// Stop transitions temporarily
			track.style.transition = 'none';


		  
			// Re-align carousel using NEW width
			updateCarousel(false); // pass "no animation"

			// FORCE layout again (important)
			forceReflow(track);

			// Restore transition
			track.style.transition = 'transform 0.6s ease-in-out';


			// IMPORTANT: re-align current slide
			/*
			if (typeof updateCarousel === 'function') {
			  updateCarousel(); // keeps slide perfectly aligned
			}
			*/
		  }, 150);
	});


  // Init
  cleanupTempClone();
  goTo(currentIndex, false);
  start();
});

        function getCookieVal(offset) {
            var endstr = document.cookie.indexOf(";", offset);
            if (endstr == -1)
                endstr = document.cookie.length;
            return unescape(document.cookie.substring(offset, endstr));
        }

        function GetCookie(name) {
            var arg = name + "=";
            var alen = arg.length;
            var clen = document.cookie.length;
            var i = 0;
            while (i < clen) {
                var j = i + alen;
                if (document.cookie.substring(i, j) == arg)
                    return getCookieVal(j);
                i = document.cookie.indexOf(" ", i) + 1;
                if (i == 0)
                    break;
            }
            return null;
        }

        function SetCookie(name, value) {
